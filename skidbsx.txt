-- ========== FLING SPECIFIC PLAYER (CORE FUNCTION) ==========
local function flingSpecificPlayer(tp)
    if isFlinging then return end
    isFlinging = true
    if not tp or not tp.Character then isFlinging = false return end
    
    local lChar = lp.Character
    local lRoot = lChar and lChar:FindFirstChild("HumanoidRootPart")
    local lHumanoid = lChar and lChar:FindFirstChild("Humanoid")
    local tChar = tp.Character
    local tRoot = tChar and tChar:FindFirstChild("HumanoidRootPart")
    
    if not lRoot or not tRoot or not lHumanoid then isFlinging = false return end

    local savedPos = lRoot.CFrame
    local sessionActive = true
    local flingCon
    flingCon = RunService.Heartbeat:Connect(function()
        if not sessionActive or not lRoot.Parent or not tRoot.Parent then
            if flingCon then flingCon:Disconnect() flingCon = nil end
            return
        end
        pcall(_shp, lHumanoid, "MoveDirectionInternal", NAN_V3)
        pcall(_shp, lRoot, "PhysicsRepRootPart", tRoot)
        lRoot.CFrame = tRoot.CFrame
        lRoot.Velocity = ZERO_V3
        lRoot.AssemblyLinearVelocity = ZERO_V3
        lRoot.AssemblyAngularVelocity = ZERO_V3
        lRoot.RotVelocity = ZERO_V3
    end)

    task.delay(1, function()
        sessionActive = false
        if flingCon then flingCon:Disconnect() flingCon = nil end
        pcall(_shp, lRoot, "PhysicsRepRootPart", nil)
        if savedPos and lRoot and lRoot.Parent then
            lRoot.CFrame = savedPos
            lRoot.Velocity = ZERO_V3
            lRoot.AssemblyLinearVelocity = ZERO_V3
            lRoot.AssemblyAngularVelocity = ZERO_V3
            lRoot.RotVelocity = ZERO_V3
        end
        isFlinging = false
        notify("Fling", "Flinged " .. tp.Name, "skull")
    end)
end

-- ========== MM2 ROLE DETECTION ==========
if isMM2 then
    local function updateMM2Roles()
        local GetPlayerData = ReplicatedStorage:FindFirstChild("GetPlayerData", true)
        if GetPlayerData and GetPlayerData:IsA("RemoteFunction") then
            local success, data = pcall(function() return GetPlayerData:InvokeServer() end)
            if success and type(data) == "table" then
                for key, val in pairs(data) do
                    local pName = type(key) == "string" and key or tostring(key)
                    if type(val) == "table" then
                        if val.Name then pName = val.Name end
                        triplet[pName] = {
                            Role = val.Role or "Unknown",
                            Dead = val.Dead or val.Killed or false
                        }
                    elseif type(val) == "string" then
                        triplet[pName] = { Role = val, Dead = false }
                    end
                end
            end
        end
    end

    local function grole(p)
        if triplet[p.Name] and triplet[p.Name].Role then return triplet[p.Name].Role end
        
        local char = p.Character
        if not char then return nil end
        if char:FindFirstChild("Murderer") or char:FindFirstChild("Knife") then return "Murderer" end
        if char:FindFirstChild("Sheriff") or char:FindFirstChild("Hero") or char:FindFirstChild("Gun") then return "Sheriff" end
        if char:FindFirstChild("Innocent") then return "Innocent" end
        
        for _, tool in ipairs(char:GetChildren()) do
            if tool:IsA("Tool") then
                local tLower = string.lower(tool.Name)
                if tLower == "knife" or tLower == "murderer" then return "Murderer" end
                if tLower == "gun" or tLower == "sheriff" then return "Sheriff" end
            end
        end
        
        local head = char:FindFirstChild("Head")
        if head then
            local billboard = head:FindFirstChildWhichIsA("BillboardGui")
            if billboard then
                local textLabel = billboard:FindFirstChildWhichIsA("TextLabel")
                if textLabel and textLabel.Text then
                    local text = string.lower(textLabel.Text)
                    if string.match(text, "murderer") then return "Murderer" end
                    if string.match(text, "sheriff") or string.match(text, "hero") then return "Sheriff" end
                end
            end
        end
        return "Unknown"
    end

    task.spawn(function()
        while true do
            updateMM2Roles()
            for _, p in ipairs(Players:GetPlayers()) do
                if p ~= lp then
                    cachedRoles[p] = { Role = grole(p), Dead = false }
                end
            end
            if not isFlinging then
                if autoFlingMurderer then
                    for p, data in pairs(cachedRoles) do
                        if data.Role == "Murderer" and not data.Dead then
                            flingSpecificPlayer(p)
                            break
                        end
                    end
                elseif autoFlingSheriff then
                    for p, data in pairs(cachedRoles) do
                        if data.Role == "Sheriff" and not data.Dead then
                            flingSpecificPlayer(p)
                            break
                        end
                    end
                end
            end
            task.wait(1)
        end
    end)
end

-- ========== UNIVERSAL FLING FUNCTION (for all toggles) ==========
local function createFlingFunction(offsetVector, offsetAngle, duration, extraSettings)
    return function(targetPlayer)
        if not targetPlayer then 
            targetPlayer = getTargetFromDropdown()
        end
        if not targetPlayer or not targetPlayer.Character then return end
        
        local lChar = lp.Character
        local lRoot = lChar and lChar:FindFirstChild("HumanoidRootPart")
        local lHumanoid = lChar and lChar:FindFirstChild("Humanoid")
        local tChar = targetPlayer.Character
        local tRoot = tChar and tChar:FindFirstChild("HumanoidRootPart")
        
        if not lRoot or not tRoot or not lHumanoid then return end
        
        local savedPos = lRoot.CFrame
        local sessionActive = true
        local CF_OFFSET = CFrame.new(offsetVector) * offsetAngle
        
        local flingCon
        flingCon = RunService.Heartbeat:Connect(function()
            if not sessionActive or not lRoot.Parent or not tRoot.Parent then
                if flingCon then flingCon:Disconnect() end
                return
            end
            pcall(_shp, lHumanoid, "MoveDirectionInternal", NAN_V3)
            pcall(_shp, lRoot, "PhysicsRepRootPart", tRoot)
            lRoot.CFrame = tRoot.CFrame * CF_OFFSET
            lRoot.Velocity = ZERO_V3
            lRoot.AssemblyLinearVelocity = ZERO_V3
            lRoot.AssemblyAngularVelocity = ZERO_V3
            lRoot.RotVelocity = ZERO_V3
            
            if extraSettings and extraSettings.heartbeat then
                extraSettings.heartbeat(lHumanoid)
            end
        end)
        
        task.delay(duration, function()
            sessionActive = false
            if flingCon then flingCon:Disconnect() end
            pcall(_shp, lRoot, "PhysicsRepRootPart", nil)
            if extraSettings and extraSettings.stop then
                extraSettings.stop()
            end
            if savedPos and lRoot and lRoot.Parent then
                lRoot.CFrame = savedPos
            end
            notify("Done", "Finished " .. (extraSettings and extraSettings.name or "fling"), "check")
        end)
    end
end

-- ========== UPDATE PLAYER LIST ==========
local function updatePlayerList()
    local playerList = {}
    if isMM2 then
        table.insert(playerList, "Murderer")
        table.insert(playerList, "Sheriff")
    end
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= lp then
            table.insert(playerList, player.Name)
        end
    end
    if #playerList == 0 then
        table.insert(playerList, "No other players")
    end
    return playerList
end

-- Bang Animation
local bangActive = false
local bangAnimTrack1, bangAnimTrack2, bangAnimLoop = nil, nil, nil

local function playBangAnimation(humanoid)
    if not humanoid then return end
    if not bangAnimTrack1 then
        local anim1 = Instance.new("Animation")
        anim1.AnimationId = "rbxassetid://215262147"
        bangAnimTrack1 = humanoid:LoadAnimation(anim1)
    end
    if not bangAnimTrack2 then
        local anim2 = Instance.new("Animation")
        anim2.AnimationId = "rbxassetid://185299570"
        bangAnimTrack2 = humanoid:LoadAnimation(anim2)
    end
    bangAnimTrack1:Play()
    task.delay(0.5, function()
        if bangAnimTrack2 then bangAnimTrack2:Play() end
    end)
end

-- Click Fling
local clickFlingEnabled = false
local clickFlingCon = nil

local function startClickFling()
    clickFlingCon = UserInputService.InputBegan:Connect(function(input, gameProcessed)
        if gameProcessed then return end
        if input.UserInputType ~= Enum.UserInputType.MouseButton1 then return end
        
        local mouse = lp:GetMouse()
        local target = mouse.Target
        if not target then return end
        
        local character = target:FindFirstAncestorWhichIsA("Model")
        if not character then return end
        local humanoid = character:FindFirstChild("Humanoid")
        if not humanoid then return end
        local targetPlayer = Players:GetPlayerFromCharacter(character)
        if not targetPlayer or targetPlayer == lp then return end
        
        flingSpecificPlayer(targetPlayer)
    end)
end

local function stopClickFling()
    if clickFlingCon then
        clickFlingCon:Disconnect()
        clickFlingCon = nil
    end
end

OPTab:Toggle({
    Title = "🖱️ CLICK FLING",
    Desc = "Click on any player to fling them",
    Icon = "mouse-pointer",
    Value = false,
    Callback = function(value)
        clickFlingEnabled = value
        if clickFlingEnabled then
            startClickFling()
            notify("Click Fling", "Click on any player to fling!", "check")
        else
            stopClickFling()
        end
    end
})

-- Touch Fling
local touchFlingEnabled = false
local touchedCon = nil
local touchFlingConnections = {}

local function startTouchFling()
    local function onTouched(hit)
        if not touchFlingEnabled then return end
        local char = hit:FindFirstAncestorWhichIsA("Model")
        if not char then return end
        local humanoid = char:FindFirstChildOfClass("Humanoid")
        if not humanoid then return end
        local targetPlayer = Players:GetPlayerFromCharacter(char)
        if not targetPlayer or targetPlayer == lp then return end
        if touchFlingConnections[targetPlayer] then return end
        
        local lChar = lp.Character
        local lHrp = lChar and lChar:FindFirstChild("HumanoidRootPart")
        local tHrp = char:FindFirstChild("HumanoidRootPart")
        if not lHrp or not tHrp then return end
        
        local sessionActive = true
        local con = RunService.Heartbeat:Connect(function()
            if not sessionActive then return end
            if not lHrp or not lHrp.Parent then
                sessionActive = false
                return
            end
            if not tHrp or not tHrp:IsDescendantOf(Workspace) then
                tHrp = char:FindFirstChild("HumanoidRootPart")
                if not tHrp then return end
            end
            pcall(_shp, lHrp, "PhysicsRepRootPart", tHrp)
            lHrp.CFrame = tHrp.CFrame
            lHrp.Velocity = ZERO_V3
            pcall(_shp, lChar.Humanoid, "MoveDirectionInternal", NAN_V3)
        end)
        
        touchFlingConnections[targetPlayer] = con
        
        task.delay(1.5, function()
            sessionActive = false
            if con then con:Disconnect() end
            touchFlingConnections[targetPlayer] = nil
            if lHrp and lHrp.Parent then
                pcall(_shp, lHrp, "PhysicsRepRootPart", nil)
            end
        end)
    end
    
    local char = lp.Character
    if char then
        local hrp = char:FindFirstChild("HumanoidRootPart")
        if hrp then touchedCon = hrp.Touched:Connect(onTouched) end
    end
    
    lp.CharacterAdded:Connect(function(newChar)
        if not touchFlingEnabled then return end
        task.wait(0.5)
        local newHrp = newChar:FindFirstChild("HumanoidRootPart")
        if newHrp then
            if touchedCon then touchedCon:Disconnect() end
            touchedCon = newHrp.Touched:Connect(onTouched)
        end
    end)
end

local function stopTouchFling()
    touchFlingEnabled = false
    if touchedCon then touchedCon:Disconnect() touchedCon = nil end
    for _, con in pairs(touchFlingConnections) do
        if con then con:Disconnect() end
    end
    touchFlingConnections = {}
    local char = lp.Character
    if char then
        local hrp = char:FindFirstChild("HumanoidRootPart")
        if hrp then pcall(_shp, hrp, "PhysicsRepRootPart", nil) end
    end
end

OPTab:Toggle({
    Title = "🤝 TOUCH FLING",
    Desc = "Fling anyone who touches you",
    Icon = "hand",
    Value = false,
    Callback = function(value)
        touchFlingEnabled = value
        if touchFlingEnabled then
            startTouchFling()
            notify("Touch Fling", "Enabled!", "check")
        else
            stopTouchFling()
        end
    end
})

-- Fling All
local flingAllActive = false
local flingAllCon = nil
local flingAllIndex = 1
local flingAllList = {}

local function startFlingAll()
    flingAllList = {}
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= lp then table.insert(flingAllList, p) end
    end
    if #flingAllList == 0 then return end
    
    flingAllIndex = 1
    flingAllActive = true
    
    local function flingNext()
        if not flingAllActive or flingAllIndex > #flingAllList then
            flingAllActive = false
            notify("Fling All", "Completed!", "check")
            return
        end
        
        local target = flingAllList[flingAllIndex]
        if target and target.Character then
            flingSpecificPlayer(target)
        end
        flingAllIndex = flingAllIndex + 1
        task.delay(1.5, flingNext)
    end
    
    flingNext()
end

OPTab:Button({
    Title = "💥 FLING ALL",
    Desc = "Fling every player in the server",
    Icon = "users",
    Callback = function()
        if flingAllActive then
            flingAllActive = false
            notify("Fling All", "Stopped!", "x")
        else
            startFlingAll()
        end
    end
})
